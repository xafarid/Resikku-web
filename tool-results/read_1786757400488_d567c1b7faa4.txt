   151→let cart = JSON.parse(localStorage.getItem('resikku_cart') || '[]');
   152→
   153→function saveCart() {
   154→  localStorage.setItem('resikku_cart', JSON.stringify(cart));
   155→  updateCartBadge();
   156→  renderCart();
   157→}

   159→function addToCart(productId, quantity = 1) {
   160→  const product = PRODUCTS.find(p => p.id === productId);
   161→  if (!product) return;
   162→  
   163→  const existing = cart.find(item => item.id === productId);
   164→  if (existing) {
   165→    existing.quantity += quantity;
   166→  } else {
   167→    cart.push({ id: productId, quantity });
   168→  }
   169→  
   170→  saveCart();
   171→  showToast(`${product.name} (${quantity}x) berhasil ditambahkan`);
   172→
   173→  // Animate cart badge
   174→  if (cartBadgeEl) {
   175→    cartBadgeEl.classList.add('bounce');
   176→    setTimeout(() => cartBadgeEl.classList.remove('bounce'), 600);
   177→  }
   178→}
   179→
   180→function removeFromCart(productId) {
   181→  cart = cart.filter(item => item.id !== productId);
   182→  saveCart();
   183→}
   184→
   185→function updateCartQuantity(productId, delta) {
   186→  const item = cart.find(i => i.id === productId);
   187→  if (!item) return;
   188→  
   189→  item.quantity += delta;
   190→  if (item.quantity <= 0) {
   191→    removeFromCart(productId);
   192→  } else {
   193→    saveCart();
   194→  }
   195→}
   196→
   197→function getCartTotal() {
   198→  return cart.reduce((total, item) => {
   199→    const product = PRODUCTS.find(p => p.id === item.id);
   200→    return total + (product ? product.price * item.quantity : 0);
   201→  }, 0);
   202→}
   203→
   204→function getCartCount() {
   205→  return cart.reduce((count, item) => count + item.quantity, 0);
   206→}

   586→function updateActiveNav() {
   587→  const scrollPos = window.scrollY + 140; // Offset for sticky navbar
   588→  const windowHeight = window.innerHeight;
   589→  const docHeight = document.documentElement.scrollHeight;
   590→
   591→  // If scrolled to near the bottom of the page, activate the last tab (Ulasan)
   592→  if (window.scrollY + windowHeight >= docHeight - 80) {
   593→    navLinks.forEach(link => {
   594→      link.classList.toggle('active', link.dataset.nav === 'testimonials');
   595→    });
   596→    return;
   597→  }
   598→
   599→  let currentId = 'hero';
   600→  sections.forEach(section => {
   601→    const top = section.offsetTop;
   602→    const height = section.offsetHeight;
   603→    if (scrollPos >= top && scrollPos < top + height) {
   604→      currentId = section.id;
   605→    }
   606→  });
   607→
   608→  navLinks.forEach(link => {
   609→    link.classList.toggle('active', link.dataset.nav === currentId);
   610→  });
   611→}

   631→// SCROLL — Reveal Animations (IntersectionObserver)
   632→// ──────────────────────────────────────────
   633→const revealObserver = new IntersectionObserver((entries) => {
   634→  entries.forEach(entry => {
   635→    if (entry.isIntersecting) {
   636→      entry.target.classList.add('visible');
   637→      revealObserver.unobserve(entry.target);
   638→    }
   639→  });
   640→}, {
   641→  threshold: 0.12,
   642→  rootMargin: '0px 0px -40px 0px'
   643→});

   645→function observeElements() {
   646→  document.querySelectorAll('.reveal, .reveal-left, .reveal-right').forEach(el => {
   647→    const rect = el.getBoundingClientRect();
   648→    if (rect.top <= (window.innerHeight || document.documentElement.clientHeight) + 50) {
   649→      el.classList.add('visible');
   650→    } else {
   651→      el.classList.add('reveal-init');
   652→      revealObserver.observe(el);
   653→    }
   654→  });
   655→}
